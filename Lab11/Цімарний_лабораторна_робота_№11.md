from math import isfinite
from time import sleep

while True:
    try:
        match_code = int(input("Введіть бажане завдання (0 - завершення програми, 2, 3, 4, 5, 6): "))
        
        match match_code:
            case 0:
                print("\nПрограма завершена.\n")
                break
            case 2:
                #----Zavdanya 2----
                num_list = []

                for i in range(1, 11):

                    if i % 2 != 0:
                        num_list.append(i)

                print(f"\nЕлементи: {num_list}\n")

            case 3:
                #----Zavdanya 3----
                num_list = []

                for i in range(1, 21):

                    if i % 2 == 0:
                        num_list.append(i)

                print(f"\nЕлементи: {num_list}\n")
            
            case 4:
                #----Zavdanya 4----
                num_list = []
                num_sum = 0

                while True:
                    try:
                        number = float(input("\nВведіть число: "))

                        if not isfinite(number):
                            raise ValueError
                        elif number == 0:
                            print("\nПрограма завершена.\n")
                            break

                        num_list.append(number)
                        num_sum += number
                        print(f"Масив: {num_list}\n"
                            f"Сума елементів: {round(num_sum, 5)}\n")

                    except ValueError:
                        print("Помилка. Введіть число!!!\n")

            case 5:
                #----Zavdanya 5----
                try:
                    seconds_value = int(input("\nВведіть кількість секунд: "))
                
                    if seconds_value <= 0:
                        print("\nВведіть ціле число > 0!!!\n")
                        continue

                    while seconds_value > 0:
                        print(f"Залишилось {seconds_value} сек.")
                        sleep(1)
                        seconds_value -= 1

                    print("Час вийшов!!!\n")

                except ValueError:
                    print("\nВведіть число!!!\n")
                    continue

            case 6:
                #----Zavdanya 6----
                list_start = -2.25          # Початок
                list_end = -2               # Кінець
                step = 0.05                 # Крок
                constant_term = 74.2        # Вільний член

                number = list_start
                while number <= list_end + step / 2:
                    x_rounded = round(number, 2)
                    y = 0.0084 * (x_rounded**(5/4) + constant_term) / (x_rounded**2 + 3.62)
                    print(f"\nЗначення: x = {x_rounded}, y = {y:.4f}\n")
                    number += step

            case _:
                print("Обраний варіант не існує!!!")
    except ValueError:
        print("Введіть ціле число!!!")
        continue

<span style="font-size: 18px">**Скрипт**</span>
```python
from math import exp

list_start = 0
list_end = 1
step = 0.1 * (list_end - list_start)
reference_value = 4.1

num_total_value = 0
num_multiplication_value = 1
x = list_start

print("\n--- Табулювання функції y = exp(-2 * x) ---\n"
      "-----------------------------\n"
      "|   x   |          y        |\n"
      "-----------------------------\n")

while x <= list_end + step/2:
    x_rounded = round(x, 3)
    y = exp(-2 * x_rounded)

    print(f"| {x:^5.2f} | {y:^17.4f} |")

    if y > reference_value:
        num_total_value += 1
        num_multiplication_value *= y
    x += step

print("-----------------------------\n"
      "\n--- Результати умовного обчислення ---\n"
       f"Умова: y > {reference_value}\n"
       f"Кількість значень, що відповідають умові: {num_total_value}")

if num_total_value <= 1:
    print(f"Добутку немає!")
else:
    print(f"Добуток цих значень y: {num_multiplication_value:.4f}")

print("--------------------------------------")
```

## 4 Висновок
<p align="justify">
&emsp;&emsp;<span style="font-size: 18px">Під час виконання лабораторної роботи я засвоїв принципи роботи циклу while у Python, його синтаксис та особливості використання. Сформував навички створення програм із повторюваними обчисленнями, які виконуються до досягнення певної умови. Для виконання першого завдання було прийнято стратегічне рішення скомпонувати всі пункти в один файл, де навігація між пунктами завдання буде відбуватись через 'match: ... case:'; при введенні 0 програма завершується. Такий підхід є дуже зручним як для програмування, так і для роботи з самою програмою. В другому завданні реалізовано консольний інтерфейс згідно прикладу, якщо використовувати дані в моєму варіанті числа, то чисел що будуть відповідати умові y > 4.1 не буде. Також до останнього елементу ряду було додано пів кроку (h/2) для запобігання помилки пов'язаною з числами з плаваючою комою, що може призвести до виходу за межі останнього елементу.
